cat DEVL LGTL > ANTP 
cat CASS DAVS MAWS ROSS > ANTC
cat BOUV GEOR SHET > SAPF

cat TRIS AMST > NSTF 
cat BECA MAGD HORN > CHIL 
cat FKMV BLAN > FKLD

TajimaSlide.sh
#!/bin/bash
cd /data2/lab2/skuas/tajima
gunzip ${1}.vcf.gz ;
vk tajima 50,000 25,000 ${1}.vcf > ${1}_Tajima_sliding  ;
gzip ${1}.vcf ;
