het_15X.sh
#!/bin/bash
source activate handsonVCF
cd /data2/lab2/skuas/heterozygosity_15X
for i in AMST BECA BOUV CASS CROZ DAVS DEVL FKMV GEOR HORN KERG LGTL MACQ MAGD MARI MAWS SHET;
do echo $i ;
vcftools --gzvcf ${i}.vcf.gz --het --out ${i}_heterozygosity ;
done


het_manual_15X.sh
#!/bin/bash
source activate handsonVCF
cd /data2/lab2/skuas/heterozygosity_15X
touch Heterozygosity_manual.txt
cat samples | while read line ;
do echo ${line} ;
bcftools view -s ${line} -O z -o ${line}.vcf.gz n58_filtered4_pruned_newID.vcf ;
Hetero=$(zcat ${line}.vcf.gz | grep -v "#" | grep "0/1" | wc -l)
Total=$(zcat ${line}.vcf.gz | grep -v "#" | wc -l)
echo -e "${line}\t${Hetero}\t${Total}" >> Heterozygosity_manual.txt
done
