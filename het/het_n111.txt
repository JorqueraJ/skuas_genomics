het.sh
#!/bin/bash
cd /data2/lab2/skuas/heterozygosity
for i in TRIS AMST CHAT MACQ KERG CROZ MARI BOUV GEOR SHET DEVL LGTL ROSS CASS DAVS MAWS HORN BECA MAGD FKMV BLAN;
do echo $i ;
vcftools --gzvcf ${i}.vcf.gz --het --out ${i}_heterozygosity ;
done


STEP3_het_manual.sh
#!/bin/bash
cd /data2/lab2/skuas/heterozygosity
touch Heterozygosity_manual.txt
cat samples | while read line ;
do echo ${line} ;
bcftools view -s ${line} -O z -o ${line}.vcf.gz n111_filtered4_pruned_newID.vcf ;
Hetero=$(zcat ${line}.vcf.gz | grep -v "#" | grep "0/1" | wc -l)
Total=$(zcat ${line}.vcf.gz | grep -v "#" | wc -l)
echo -e "${line}\t${Hetero}\t${Total}" >> Heterozygosity_manual.txt
done
