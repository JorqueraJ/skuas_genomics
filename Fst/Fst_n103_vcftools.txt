#GLOBAL FST
1_fst_populations.sh
#!/bin/bash
source activate handsonVCF
cd /data2/lab2/skuas/Fst
# Define your populations in an array
populations=(TRIS AMST MACQ KERG CROZ MARI BOUV GEOR SHET DEVL LGTL ROSS CASS DAVS MAWS BECA MAGD FKMV)

# Loop through all pairs of populations
for ((i=0; i<${#populations[@]}-1; i++)); do
  for ((j=i+1; j<${#populations[@]}; j++)); do
    pop1=${populations[$i]}
    pop2=${populations[$j]}
    
    # Run vcftools Fst calculation
    vcftools --gzvcf n103_filtered4_pruned_newID.vcf.gz \
    --weir-fst-pop $pop1 \
    --weir-fst-pop $pop2 \
    --out ./Fst_${pop1}_${pop2}
  done
done

#### Average

#!/bin/bash

# Change to the directory with the files
cd /data2/lab2/skuas/Fst
# Define the list of files
files="
# Loop through each file
for file in $files; do
    # Use awk to sum the values and count the number of valid (non-nan) entries
    sum=$(awk '$3 != "-nan" {sum += $3} END {print sum}' "$file")
    count=$(awk '$3 != "-nan" {count++} END {print count}' "$file")

    # Calculate the average
    if [ "$count" -gt 0 ]; then
        average=$(echo "$sum / $count" | bc -l)
        echo "File: $file"
        echo "Sum of WEIR_AND_COCKERHAM_FST: $sum"
        echo "Average of WEIR_AND_COCKERHAM_FST: $average"
        echo "-----------------------------------"
    else
        echo "File: $file"
        echo "No valid entries found."
        echo "-----------------------------------"
    fi
done
